# Developer Menu (`pixi run dev`) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make `pixi run dev` open an interactive Developer menu (launch dev mode, regenerate one workflow, open repo in VSCode, regenerate all workflows), reusing existing actions and a shared `_launch_vscode` helper.

**Architecture:** Extract `_launch_vscode` from `action_open_in_vscode` in `src/assist/workspace/setup.py`; add three new dev actions + `print_dev_menu`/`run_dev_menu`; add a `dev` subcommand to `cli.py`; repoint the `dev` pixi task to the menu.

**Tech Stack:** Python 3.11, `unittest` + `pytest`, `subprocess`, `shutil`, existing `assist.workspace` package.

## Global Constraints

- Edits confined to `src/assist/workspace/setup.py`, `src/assist/workspace/cli.py`, `pyproject.toml`, and `tests/test_workspace_setup.py`. No changes to `service.py`, `bridge.py`, `make_notebooks.py`.

- Tests: `unittest.TestCase` in `tests/test_workspace_setup.py`, run with `pytest`; inject `print_func`/`input_func`, patch with `unittest.mock`.

- Run the full module after each task: `pytest tests/test_workspace_setup.py -v`.

```python
def _launch_vscode(
    target_root,
    *,
    extra_paths=(),
    print_func=print,
    input_func=input,
    which=shutil.which,
    launcher=subprocess.Popen,
    candidates=None,
    max_retries: int = 3,
) -> Path | None:
    code_path = _find_code_executable(which=which, candidates=candidates)
    attempts = 0
    while code_path is None and attempts < max_retries:
        attempts += 1
        print_func("VSCode 'code' command not found.")
        print_func(
            "In VSCode: press Cmd/Ctrl+Shift+P and run "
            "\"Shell Command: Install 'code' command in PATH\"."
        )
        answer = input_func(
            "Press Enter when done to retry, or type 's' to skip: "
        ).strip().lower()
        if answer in {"s", "skip"}:
            break
        code_path = _find_code_executable(which=which, candidates=candidates)

    if code_path is None:
        print_func(f"Open this folder in VSCode manually: {target_root}")
        return None

    args = [code_path, str(target_root), *extra_paths]
    try:
        launcher(args)
    except (OSError, FileNotFoundError) as exc:
        print_func(f"Error: failed to open VSCode: {exc}")
        print_func(f"Open this folder in VSCode manually: {target_root}")
        return None

    print_func(f"Opening {target_root} in VSCode…")
    return target_root


def action_open_in_vscode(
    state: SetupState,
    *,
    print_func=print,
    input_func=input,
    which=shutil.which,
    launcher=subprocess.Popen,
    candidates=None,
    max_retries: int = 3,
) -> Path | None:
    if not require_current_project(state, print_func=print_func):
        return None

    workspace_root = require_workspace_root(state)
    project_root = bridge.get_project_dir(workspace_root, state.current_project)

    if nhm_notebooks_need_generation(state):
        print_func("Generating NHM notebooks before opening VSCode…")
        generate_nhm_notebooks(state, print_func=print_func)

    first_nb = project_root / "notebooks" / "nhm" / "0_workspace_setup.ipynb"
    extra_paths = [str(first_nb)] if first_nb.exists() else []
    return _launch_vscode(
        project_root,
        extra_paths=extra_paths,
        print_func=print_func,
        input_func=input_func,
        which=which,
        launcher=launcher,
        candidates=candidates,
        max_retries=max_retries,
    )
```

```python
def _prompt_workflow_choice(*, print_func=print, input_func=input) -> str | None:
    workflows = list(bridge.WORKFLOW_NAMES)
    print_func("Choose a workflow:")
    print_func("0. Cancel")
    for idx, name in enumerate(workflows, start=1):
        print_func(f"{idx}. {name}")
    choice = prompt_menu_choice(
        len(workflows), input_func=input_func, print_func=print_func
    )
    if choice == 0:
        return None
    return workflows[choice - 1]


def action_regenerate_workflow(
    state: SetupState,
    *,
    print_func=print,
    input_func=input,
) -> list[Path] | None:
    workflow = _prompt_workflow_choice(print_func=print_func, input_func=input_func)
    if workflow is None:
        print_func("Regeneration cancelled.")
        return None
    created = notebook_builder.convert_workflow(workflow)
    notebook_dir = bridge.get_workflow_notebooks_dir(workflow)
    print_func(f"Regenerated {workflow} notebooks in {notebook_dir}")
    return created


def action_regenerate_all_workflows(
    state: SetupState,
    *,
    print_func=print,
) -> list[str]:
    names = list(bridge.WORKFLOW_NAMES)
    for name in names:
        notebook_builder.convert_workflow(name)
        print_func(f"Regenerated {name} notebooks.")
    return names
```

Note: `action_launch_dev_mode` currently prints its own workflow menu inline. Leave it unchanged for this plan (it already works and is tested); `_prompt_workflow_choice` is used only by the new regenerate action.

```python
def print_dev_menu(state: SetupState, *, print_func=print) -> None:
    print_func("")
    print_func("Developer menu")
    print_func("")
    print_func(f"Repo root: {state.repo_root}")
    print_func("")
    print_func("  1. Launch dev mode (regenerate + JupyterLab)")
    print_func("  2. Regenerate notebooks (no launch)")
    print_func("  3. Open repo in VSCode")
    print_func("  4. Regenerate ALL workflows")
    print_func("  0. Exit")


def run_dev_menu(
    *,
    repo_root: str | Path | None = None,
    print_func=print,
    input_func=input,
) -> int:
    resolved_repo_root = (
        bridge.resolve_repo_root()
        if repo_root is None
        else Path(repo_root).expanduser().resolve()
    )
    state = SetupState(repo_root=resolved_repo_root, workspace_root=None)

    try:
        while True:
            print_dev_menu(state, print_func=print_func)
            choice = prompt_menu_choice(4, input_func=input_func, print_func=print_func)
            if choice == 0:
                print_func("Exiting developer menu.")
                return 0
            try:
                if choice == 1:
                    action_launch_dev_mode(
                        state, print_func=print_func, input_func=input_func
                    )
                elif choice == 2:
                    action_regenerate_workflow(
                        state, print_func=print_func, input_func=input_func
                    )
                elif choice == 3:
                    action_open_repo_in_vscode(
                        state, print_func=print_func, input_func=input_func
                    )
                elif choice == 4:
                    action_regenerate_all_workflows(state, print_func=print_func)
            except (FileNotFoundError, NotADirectoryError, ValueError, OSError) as exc:
                print_func(f"Error: {exc}")
    except KeyboardInterrupt:
        print_func("\nDeveloper menu cancelled.")
        return 130
```
